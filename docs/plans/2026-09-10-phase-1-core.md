# Bearingkit Phase 1 (core) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Bearingkit core (`core/AGENTS.md`, `bk-protocol`, the ten protocol skills, `detect-stack`, the `stack-profile` hook, both host adapters, the dev installer) and pass the Phase 1 gate: 10/10 activation on the six Phase 1 intents on both hosts and at most 5,000 fixed tokens, measured in an isolated host profile.

**Architecture:** One `core/` folder is the only source; Claude Code sees it through directory junctions created by `bin/bearingkit.cjs install --dev`, Antigravity through a plugin folder that links back to `core/`. Hooks are Node scripts that detect the host from the payload shape and share one state file per session under `~/.bearingkit/state/`. Skills are Agent-Skills-spec markdown with descriptions that act as the router; `core/AGENTS.md` is the always-on protocol.

**Tech Stack:** Node 20+ (owner machine: Node 25), `node:test` for unit tests, no runtime dependencies. Markdown with LF endings and no BOM. Spec: `docs/specs/2026-09-10-bearingkit-v1-design.md` (v1.4).

**Conventions for every task:** commit after each task with a conventional message; every claim in a commit or report is backed by pasted command output; files under `_build/` are gitignored scratch; nothing under `~/.claude` or `~/.gemini` is written by any task except Task 12, which uses an isolated config directory, and Task 13, which needs the owner's approval in the turn it runs.

- [ ] **Step 3: Implement `scripts/detect-stack.cjs`**

Structure (one function per manifest, one merge step):
```js
'use strict';
const fs = require('node:fs');
const path = require('node:path');

const HOT_DEFAULT = ['**/auth/**', '**/session*/**', '**/middleware/**', '**/migrations/**', '**/payments/**', '**/billing/**', '**/upload*/**', '**/tenant*/**', '**/roles/**', '**/permissions/**'];
const exists = (d, f) => fs.existsSync(path.join(d, f));
const readJson = (d, f) => { try { return JSON.parse(fs.readFileSync(path.join(d, f), 'utf8')); } catch { return null; } };
const major = (v) => { const m = String(v || '').match(/(\d+)/); return m ? Number(m[1]) : null; };

function lockMajor(dir, name) {
  for (const f of ['pnpm-lock.yaml', 'package-lock.json', 'yarn.lock']) {
    if (!exists(dir, f)) continue;
    const txt = fs.readFileSync(path.join(dir, f), 'utf8');
    const m = txt.match(new RegExp(`${name.replace('/', '\\/')}@(\\d+)`)) || txt.match(new RegExp(`"node_modules/${name}":\\s*{[^}]*"version":\\s*"(\\d+)`));
    if (m) return Number(m[1]);
  }
  return null;
}

function node(dir) {
  const pkg = readJson(dir, 'package.json'); if (!pkg) return null;
  const pm = exists(dir, 'pnpm-lock.yaml') ? 'pnpm' : exists(dir, 'yarn.lock') ? 'yarn' : exists(dir, 'bun.lockb') ? 'bun' : 'npm';
  const deps = { ...(pkg.dependencies || {}), ...(pkg.devDependencies || {}) };
  const known = ['next', 'react', 'vue', 'nuxt', 'electron', 'express', '@nestjs/core', 'prisma', 'vitest', 'jest', '@playwright/test', 'typescript'];
  const frameworks = known.filter((k) => deps[k]).map((k) => ({ name: k === '@nestjs/core' ? 'nest' : k === '@playwright/test' ? 'playwright' : k, major: lockMajor(dir, k) ?? major(deps[k]) }));
  const run = (s) => (pm === 'npm' ? `npm run ${s}` : `${pm} ${s}`);
  const commands = {};
  if (pkg.scripts?.test) commands.test = run('test');
  if (exists(dir, 'tsconfig.json')) commands.typecheck = `${pm === 'npm' ? 'npx' : pm + ' exec'} tsc --noEmit`;
  if (pkg.scripts?.lint) commands.lint = run('lint');
  if (pkg.scripts?.build) commands.build = run('build');
  const hot = [...HOT_DEFAULT]; if (deps.prisma || deps['@prisma/client']) hot.push('**/prisma/schema.prisma');
  return { languages: [exists(dir, 'tsconfig.json') ? 'typescript' : 'javascript'], frameworks: frameworks.filter((f) => f.name !== 'typescript'), packageManager: pm, commands,
    guardrails: [commands.test, commands.typecheck].filter(Boolean), sourceExtensions: ['.ts', '.tsx', '.js', '.jsx', '.mjs', '.cjs'], hotPathGlobs: hot,
    versionCard: frameworks.filter((f) => f.major).map((f) => `${cap(f.name)} ${f.major}`).join(', ') };
}
// php(), python(), gradle(), cmake(), dotnet(), go(), rust() follow the same shape; each returns null when its manifest is absent.
const cap = (s) => s.charAt(0).toUpperCase() + s.slice(1);

function missingBinaries(commands, envPath) {
  const dirs = (envPath ?? process.env.PATH ?? '').split(path.delimiter).filter(Boolean);
  const exts = process.platform === 'win32' ? (process.env.PATHEXT || '.EXE;.CMD;.BAT').split(';') : [''];
  const found = (bin) => dirs.some((d) => exts.some((e) => fs.existsSync(path.join(d, bin + e))));
  const bins = [...new Set(Object.values(commands).map((c) => c.split(' ')[0]).filter((b) => !b.includes('/')))];
  return bins.filter((b) => !found(b));
}

function detect(dir = process.cwd(), opts = {}) {
  const parts = [node, php, python, gradle, cmake, dotnet, go, rust].map((f) => f(dir)).filter(Boolean);
  if (!parts.length) throw new Error('no-manifest');
  const p = parts[0]; // first detected manifest wins; others contribute languages
  for (const extra of parts.slice(1)) p.languages.push(...extra.languages);
  return { schema: 1, root: dir.replace(/\\/g, '/'), vcs: exists(dir, '.git') ? 'git' : 'none', ...p, parity: { missingBinaries: missingBinaries(p.commands, opts.path) } };
}

if (require.main === module) {
  try { process.stdout.write(JSON.stringify(detect(process.argv[2] || process.cwd()), null, 2) + '\n'); }
  catch (e) { process.stdout.write(JSON.stringify({ error: e.message }) + '\n'); process.exit(2); }
}
module.exports = { detect };
```

const DEFAULTS = () => ({ schema: 1, stackHash: '', lastInjectionHash: '', branch: '', dirty: false, activePlan: null, codeFilesChanged: [], guardrailRuns: [], hotPathTouched: false, independentReview: null, tempBypassMarkers: [], handoffPresent: false });

class State {
  constructor({ host, sessionId, cwd, baseDir }) {
    this.dir = baseDir || path.join(os.homedir(), '.bearingkit', 'state');
    fs.mkdirSync(this.dir, { recursive: true });
    this.file = path.join(this.dir, `${host}-${sessionId}.json`);
    if (!fs.existsSync(this.file)) this._write({ ...DEFAULTS(), host, sessionId, cwd: cwd.replace(/\\/g, '/'), updatedAt: new Date().toISOString() });
  }
  read() { return JSON.parse(fs.readFileSync(this.file, 'utf8')); }
  update(patch) { const d = { ...this.read(), ...patch, updatedAt: new Date().toISOString() }; this._write(d); return d; }
  _write(d) { const tmp = this.file + '.tmp'; fs.writeFileSync(tmp, JSON.stringify(d, null, 2)); fs.renameSync(tmp, this.file); }
  prune(maxAgeDays = 7) {
    const cutoff = Date.now() - maxAgeDays * 86400000;
    for (const f of fs.readdirSync(this.dir)) { const p = path.join(this.dir, f); if (fs.statSync(p).mtimeMs < cutoff) fs.rmSync(p); }
  }
}
module.exports = { State };

function gitInfo(cwd) {
  try { const branch = execFileSync('git', ['rev-parse', '--abbrev-ref', 'HEAD'], { cwd, stdio: ['ignore', 'pipe', 'ignore'] }).toString().trim();
    const dirtyCount = execFileSync('git', ['status', '--porcelain'], { cwd, stdio: ['ignore', 'pipe', 'ignore'] }).toString().split('\n').filter(Boolean).length;
    return { branch, dirtyCount }; } catch { return null; }
}
function activePlan(cwd, branch, state) {
  if (state.activePlan) return state.activePlan;
  const slug = (branch || '').split('/').pop().replace(/^\d+-/, ''); if (!slug) return null;
  const dir = path.join(cwd, 'plans'); if (!fs.existsSync(dir)) return null;
  const hit = fs.readdirSync(dir).find((d) => d.includes(slug)); return hit ? { path: `plans/${hit}`, phase: null } : null;
}
function latestHandoff(cwd) {
  const dir = path.join(cwd, 'docs', 'handoff'); if (!fs.existsSync(dir)) return null;
  const f = fs.readdirSync(dir).filter((x) => x.endsWith('.md')).sort().pop(); if (!f) return null;
  return Date.now() - fs.statSync(path.join(dir, f)).mtimeMs < 14 * 86400000 ? `docs/handoff/${f}` : null;
}
function buildBlock(p, git, plan, handoff, hot) {
  const l1 = `[bearingkit] stack: ${p.languages.join(', ')} · ${p.versionCard || 'no framework detected'} · ${p.packageManager || 'no package manager'}`;
  const l2 = `guardrails: ${p.guardrails.length ? p.guardrails.join('; ') : 'none detected'}`;
  const l3 = git ? `branch: ${git.branch}${git.dirtyCount ? ` (dirty: ${git.dirtyCount} files)` : ''}${plan ? ` · plan: ${plan.path}${plan.phase ? ` phase ${plan.phase}` : ''}` : ''}` : 'no git';
  const l4 = `hot path touched: ${hot ? 'yes' : 'no'}${handoff ? ` · handoff: ${handoff}` : ''}`;
  return [l1, l2, l3, l4].join('\n');
}
async function run(payload, deps = {}) {
  const e = parse(payload);
  if (e.isSubagent || !['session-start', 'prompt', 'post-compact', 'pre-invocation'].includes(e.event)) return {};
  const state = new State({ host: e.host, sessionId: e.sessionId, cwd: e.cwd, baseDir: deps.baseDir });
  if (e.event === 'session-start') state.prune();
  if (e.event === 'post-compact') state.update({ lastInjectionHash: '' });
  let profile; try { profile = (deps.detect || detect)(e.cwd); } catch { profile = { languages: [], guardrails: [] }; }
  const git = (deps.git || gitInfo)(e.cwd);
  const s = state.read();

  const block = buildBlock(profile, git, plan, handoff, s.hotPathTouched);
  const h = sha(block);
  if (h === s.lastInjectionHash) return {};
  state.update({ lastInjectionHash: h, stackHash: sha(JSON.stringify(profile)), branch: git?.branch || '', dirty: Boolean(git?.dirtyCount), activePlan: plan, handoffPresent: Boolean(handoff) });
  return format(e.host, e.event, { inject: block });
}
